! =====================================================================
! R1 - Cisco 2911 - Integrador II (IP2-28)
! Router-on-a-stick + NAT/PAT + DHCP + ACL + SSH + SNMP
! Reemplazar TODO lo marcado CAMBIAR- antes de usar en equipo real.
! Pegar en: enable -> configure terminal
! =====================================================================
hostname R1
no ip domain-lookup
ip domain-name integrador2.local
service password-encryption
enable secret CAMBIAR-ENABLE-SECRET
username admin privilege 15 secret CAMBIAR-ADMIN-SECRET
banner motd ^Acceso restringido - Integrador II - Solo personal autorizado^
!
! ---------------- WAN (salida a Internet) ----------------
! Laboratorio real: la universidad entrega la IP por DHCP.
! Packet Tracer: se usa la IP fija 172.16.0.2/30 hacia el router ISP simulado.
interface GigabitEthernet0/0
 description WAN - Red de la universidad
 ip address dhcp
 ! (Packet Tracer) ip address 172.16.0.2 255.255.255.252
 ip nat outside
 no shutdown
!
! ---------------- Trunk hacia SW1 ----------------
interface GigabitEthernet0/1
 description Trunk hacia SW1 Gi0/1
 no ip address
 no shutdown
!
interface GigabitEthernet0/1.10
 description VLAN 10 - ADMINISTRACION
 encapsulation dot1Q 10
 ip address 10.10.10.1 255.255.255.0
 ip nat inside
!
interface GigabitEthernet0/1.20
 description VLAN 20 - USUARIOS
 encapsulation dot1Q 20
 ip address 10.10.20.1 255.255.255.0
 ip nat inside
 ip access-group ACL-USUARIOS-IN in
!
interface GigabitEthernet0/1.30
 description VLAN 30 - SERVIDORES
 encapsulation dot1Q 30
 ip address 10.10.30.1 255.255.255.0
 ip nat inside
!
interface GigabitEthernet0/1.99
 description VLAN 99 - GESTION
 encapsulation dot1Q 99
 ip address 10.10.99.1 255.255.255.0
 ip nat inside
!
interface GigabitEthernet0/1.999
 description VLAN 999 - NATIVA SIN USO
 encapsulation dot1Q 999 native
!
! ---------------- DHCP (solo usuarios) ----------------
ip dhcp excluded-address 10.10.20.1 10.10.20.99
ip dhcp excluded-address 10.10.20.201 10.10.20.254
ip dhcp pool VLAN20-USUARIOS
 network 10.10.20.0 255.255.255.0
 default-router 10.10.20.1
 dns-server 8.8.8.8
!
! ---------------- ACL: usuarios no administran la infraestructura ----------------
ip access-list extended ACL-USUARIOS-IN
 remark DHCP hacia el router
 permit udp any any eq bootps
 remark Ping al gateway (diagnostico basico)
 permit icmp 10.10.20.0 0.0.0.255 host 10.10.20.1
 remark Unico servicio permitido: la aplicacion web
 permit tcp 10.10.20.0 0.0.0.255 host 10.10.30.11 eq www
 permit tcp 10.10.20.0 0.0.0.255 host 10.10.30.11 eq 443
 remark Respuestas a conexiones iniciadas por Administracion (soporte remoto)
 permit tcp 10.10.20.0 0.0.0.255 10.10.10.0 0.0.0.255 established
 permit icmp 10.10.20.0 0.0.0.255 10.10.10.0 0.0.0.255 echo-reply
 remark Bloqueo de redes internas
 deny ip 10.10.20.0 0.0.0.255 10.10.10.0 0.0.0.255
 deny ip 10.10.20.0 0.0.0.255 10.10.30.0 0.0.0.255
 deny ip 10.10.20.0 0.0.0.255 10.10.99.0 0.0.0.255
 remark Bloqueo de acceso al router por sus otras IP
 deny ip 10.10.20.0 0.0.0.255 host 10.10.10.1
 deny ip 10.10.20.0 0.0.0.255 host 10.10.30.1
 deny ip 10.10.20.0 0.0.0.255 host 10.10.99.1
 remark Internet
 permit ip 10.10.20.0 0.0.0.255 any
 exit
!
! ---------------- NAT/PAT ----------------
ip access-list standard ACL-NAT-INTERNAS
 permit 10.10.10.0 0.0.0.255
 permit 10.10.20.0 0.0.0.255
 permit 10.10.30.0 0.0.0.255
 permit 10.10.99.0 0.0.0.255
 exit
ip nat inside source list ACL-NAT-INTERNAS interface GigabitEthernet0/0 overload
!
! Packet Tracer: ruta por defecto hacia el ISP simulado.
! Laboratorio real: la ruta por defecto llega por DHCP en Gi0/0 (no hace falta).
! ip route 0.0.0.0 0.0.0.0 172.16.0.1
!
! ---------------- Acceso administrativo: solo SSH desde VLAN 10 ----------------
ip access-list standard ACL-SSH-ADMIN
 permit 10.10.10.0 0.0.0.255
 exit
crypto key generate rsa general-keys modulus 2048
ip ssh version 2
line console 0
 login local
 exec-timeout 10 0
 logging synchronous
line vty 0 4
 login local
 transport input ssh
 access-class ACL-SSH-ADMIN in
 exec-timeout 10 0
!
! ---------------- SNMP: solo lectura y solo desde Zabbix ----------------
ip access-list standard ACL-SNMP-ZABBIX
 permit host 10.10.30.12
 exit
snmp-server community CAMBIAR-SNMP-COMMUNITY RO ACL-SNMP-ZABBIX
! (No soportados en Packet Tracer; sí en IOS real)
snmp-server location Laboratorio UTN - Rack Integrador II
snmp-server contact Equipo Integrador II
! Laboratorio real (si la imagen IOS lo soporta), preferir SNMPv3:
! snmp-server group ZABBIX-RO v3 priv access ACL-SNMP-ZABBIX
! snmp-server user zabbix ZABBIX-RO v3 auth sha CAMBIAR-AUTH priv aes 128 CAMBIAR-PRIV
!
end
write memory
